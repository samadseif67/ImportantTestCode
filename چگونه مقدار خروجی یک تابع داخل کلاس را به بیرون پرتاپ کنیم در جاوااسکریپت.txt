class MyClass extends EventTarget {

    constructor() {
        super();
    }

    start() {
        // فراخوانی داخلی تابع
        this.myFunction();
    }

    myFunction() {
        const result = "خروجی تابع داخلی";

        // ارسال خروجی به بیرون کلاس
        this.dispatchEvent(
            new CustomEvent("myFunctionCompleted", {
                detail: result
            })
        );
    }
}


-------------------------------------------------------------------------------------------

// استفاده از کلاس در بیرون
const obj = new MyClass();

// دریافت خروجی هر بار که تابع داخلی اجرا شود
obj.addEventListener("myFunctionCompleted", (event) => {
    console.log("خروجی دریافتی:", event.detail);
});

// اجرای تابع اصلی
obj.start();

-----------------------------------------------------------------------------------------------
این دستور رویدادی را که ساخته‌اید، ارسال می‌کند.  this.dispatchEvent

این دستور یک رویداد جدید می‌سازد؛ مثل یک پیام که قرار است به بیرون ارسال شود. new CustomEvent
"myFunctionCompleted": نام رویداد است. شما نام آن را تعیین می‌کنید و بیرون کلاس با همین نام به آن گوش می‌دهید.

detail: result: اطلاعاتی است که می‌خواهید همراه رویداد ارسال کنید. در این مثال، مقدار متغیر result ارسال می‌شود.


نکته مهم: برای اینکه این روش کار کند، کلاس باید از EventTarget ارث‌بری کند یا شیء دیگری داشته باشید که متد dispatchEvent را پشتیبانی کند.
